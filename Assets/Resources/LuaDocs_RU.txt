PCOS Lua
========
Язык внутри системы. Канал: t.me/orangePCsimu

РЕДАКТОР
--------
Слева код, справа подсказки. | в шаблоне — куда встанет курсор.
Run — превью в системном окне (нужно для ui.*).
Save — исходник .lua / .txt.
Compile — пишет Name.exe на рабочий стол (ярлык). Запуск двойным кликом.
Pick icon — иконка из .pic (или любой файл-картинка в base64).
Имя .exe — поле Compile Name в инспекторе редактора.

Документация: отдельная кнопка, OnClick = Docs (или OpenDocs / ShowDocs),
цель = это окно Lua Editor на сцене, не префаб из Project.

ПОДСВЕТКА СИНТАКСИСА
--------------------
На объект Lua Editor повесь LuaSyntaxHighlight:
  Code = InputField с кодом
  Overlay = Text-копия того же поля (тот же Rect, Rich Text ВКЛ, Raycast ВЫКЛ)
Цвет текста у InputField почти прозрачный — виден только overlay.
Цвета keyword / string / comment / number настраиваются в инспекторе.

КОМПИЛЯЦИЯ / .EXE
-----------------
Compile пакует JSON { pcos: lua-app, name, icon, script } в Name.exe.
Иконка на рабочем столе — из Pick Icon, если выбирал.
Несколько скомпилированных окон могут быть открыты сразу.

ИНТЕРФЕЙС (системный угловатый PCOS)
------------------------------------
Координаты: левый верх, Y вниз, пиксели.

ui.title("Имя")
ui.size(w, h)
ui.isdraggable(true/false)           -- также ui.draggable
ui.ismaximable(true/false)           -- также ui.ismaximizable / ui.maximizable
ui.label(text, x, y, w, h)           -> id
ui.button(text, x, y, w, h, fn)      -> id
ui.input(text, x, y, w, h)           -> id
ui.panel(x, y, w, h)                 -> id
ui.rect(x, y, w, h, r, g, b, a)      -> id
ui.image(path, x, y, w, h)           -> id  (.pic / файл в base64)
ui.slider(x, y, w, h, min, max)      -> id
ui.toggle(text, x, y, w, h, on)      -> id
ui.get(id)  ui.set(id, text)
ui.color(id, r, g, b, a)
ui.value(id)     -- слайдер
ui.checked(id)   -- галка
ui.remove(id)  ui.clear()
ui.systemstyle()
ui.style({
  window={r,g,b}, panel={r,g,b},
  button={r,g,b}, buttonText={r,g,b},
  text={r,g,b}, input={r,g,b}, inputText={r,g,b},
  fontSize=14
})
Цвет — таблица 0..1 {r,g,b} или {r,g,b,a}.

Пример:
  ui.title("Счётчик")
  ui.size(320, 180)
  local n = 0
  local lab = ui.label("0", 20, 20, 200, 24)
  ui.button("Клик", 20, 60, 100, 28, function()
    n = n + 1
    ui.set(lab, tostring(n))
  end)
